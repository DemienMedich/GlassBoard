cmake_minimum_required(VERSION 3.21)
project(GlassBoard VERSION 1.2.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets)
qt_standard_project_setup()
configure_file(assets/GlassBoard.rc.in ${CMAKE_CURRENT_BINARY_DIR}/GlassBoard.rc @ONLY)
qt_add_executable(GlassBoard WIN32 main.cpp ${CMAKE_CURRENT_BINARY_DIR}/GlassBoard.rc)
target_link_libraries(GlassBoard PRIVATE Qt6::Widgets user32)
target_compile_definitions(GlassBoard PRIVATE UNICODE _UNICODE GLASSBOARD_VERSION="${PROJECT_VERSION}")
